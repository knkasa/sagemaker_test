Instruction to use your own image in sagemaker.

- Build Your Custom Docker Image in local.
===========================================
FROM python:3.9-slim

# Install necessary dependencies
RUN pip install --no-cache-dir numpy pandas sagemaker scikit-learn

# Add training script
COPY train.py /opt/ml/code/train.py

# Set the training script as the entry point
ENV SAGEMAKER_PROGRAM train.py
WORKDIR /opt/ml/code
============================================

- Create a python script for the image. train.py
=======================================================================
import os
import pandas as pd
from sklearn.linear_model import LinearRegression
from sklearn.externals import joblib

# Define paths
input_path = "/opt/ml/input/data/train"
output_path = "/opt/ml/model"

# Load training data
data = pd.read_csv(os.path.join(input_path, "train.csv"), header=None)
X = data.iloc[:, :-1]  # Features
y = data.iloc[:, -1]   # Target

# Train the model
model = LinearRegression()
model.fit(X, y)

# Save the model
os.makedirs(output_path, exist_ok=True)
joblib.dump(model, os.path.join(output_path, "model.joblib"))
========================================================================
